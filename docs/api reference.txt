-- ============================================================
-- ENTITY API — Collision / Trigger / Wire subscriptions (17 methods)
-- ============================================================

function DemoCollisions()
    local e = Entity(inputs.entity.target)
    if e._nil then return end

    -- === Collision subscriptions ===
    -- Callback fires BEFORE OnTick (in pre-execute hook)

    local sub1 = e:subscribeCollisionEnter(function(other, self, nx, ny)
        -- other: Entity that collided with us
        -- self: Entity that owns the subscription
        -- nx, ny: collision normal
    end)

    local sub2 = e:subscribeCollisionExit(function(other, self)
        -- End of collision (no normal)
    end)

    local sub3 = e:subscribeCollisionStay(function(other, self)
        -- Every tick while objects touch
    end)

    -- === Trigger subscriptions ===
    -- Triggers have NO normal (no nx, ny)

    local sub4 = e:subscribeTriggerEnter(function(other, self)
        -- Enter trigger zone
    end)

    local sub5 = e:subscribeTriggerExit(function(other, self)
        -- Exit trigger zone
    end)

    local sub6 = e:subscribeTriggerStay(function(other, self)
        -- Every tick while inside trigger zone
    end)

    -- === Unsubscribe (specific subscription by subId) ===
    e:unsubscribeCollisionEnter(sub1)
    e:unsubscribeCollisionExit(sub2)
    e:unsubscribeCollisionStay(sub3)
    e:unsubscribeTriggerEnter(sub4)
    e:unsubscribeTriggerExit(sub5)
    e:unsubscribeTriggerStay(sub6)

    -- === Unsubscribe every subscription of this entity ===
    e:unsubscribeAll()

    -- === Wire (in/out) subscriptions on chips ===
    -- Fires when somebody connects/disconnects a wire to a gate of this chip.
    local sw1 = e:subscribeWireConnected(function(selfId, inputKey, outputEntityId, outputKey)
        -- selfId: id of the chip receiving the wire
        -- inputKey: gate name on this chip that got connected
        -- outputEntityId: id of the chip on the other end
        -- outputKey: gate name on the other chip
    end)

    local sw2 = e:subscribeWireDisconnected(function(selfId, inputKey)
        -- selfId: id of the chip
        -- inputKey: gate name that got disconnected
    end)

    e:unsubscribeWireConnected(sw1)
    e:unsubscribeWireDisconnected(sw2)
end

-- ============================================================
-- input — input (pointer / multi-touch / gestures / keyboard, 31 methods)
-- ============================================================
-- Mouse and touch are unified through LeanTouch — same API works on both
-- desktop (mouse) and mobile (touch). Mouse counts as one finger.

function DemoInput()
    -- ==== Pointer (single, "primary" finger / mouse) ====
    local down = input.pointerDown()           -- → 1/0 (pressed)
    local up = input.pointerUp()               -- → 1/0 (released this frame)
    local wx, wy = input.pointerPos()          -- → world coordinates
    local sx, sy = input.pointerScreenPos()    -- → screen pixels
    local dx, dy = input.pointerDelta()        -- → movement delta this frame

    -- Raycast — first object under pointer
    local hitId = input.pointerRaycast()       -- → entityId (0 = nothing hit)

    -- Raycast — all objects under pointer
    local allHits = input.pointerRaycastAll()  -- → array of entityId (unique)
    for i = 1, #allHits do
        local e = Entity(allHits[i])
        print("hit: " .. e:getName())
    end

    -- UI filter
    local overUI = input.isOverUI()            -- → 1/0 (pointer over UI?)
    local downFiltered = input.pointerDownFiltered()  -- → 1/0 (pressed, not started over UI)
    local upFiltered = input.pointerUpFiltered()      -- → 1/0 (released, not started over UI)

    -- ==== Multi-touch (per-finger, 1-based index) ====
    -- Index is into LeanTouch.Fingers — order ≈ touch order, stable within a frame.
    -- Use input.touchId(i) for stable hardware ID across frames.
    local count = input.touchCount()           -- → number of active fingers

    if count >= 1 then
        local set     = input.touchSet(1)              -- → 1/0 (currently touching)
        local tdown   = input.touchDown(1)             -- → 1/0 (pressed THIS frame)
        local tup     = input.touchUp(1)               -- → 1/0 (released THIS frame)
        local age     = input.touchAge(1)              -- → seconds since touch start
        local id      = input.touchId(1)               -- → stable hardware id

        local tx, ty   = input.touchPos(1)             -- → world coordinates
        local sx2, sy2 = input.touchScreenPos(1)       -- → current screen pixels
        local stx, sty = input.touchStartScreenPos(1)  -- → screen position at touch start
        local tdx, tdy = input.touchDelta(1)           -- → delta this frame
        local sdx, sdy = input.touchSwipeDelta(1)      -- → total delta since touch start

        local tap     = input.touchTap(1)              -- → 1 only on tap event frame
        local tapN    = input.touchTapCount(1)         -- → N consecutive taps (double-tap = 2)
        local swipe   = input.touchSwipe(1)            -- → 1 only on swipe event frame

        local overUi    = input.touchIsOverUI(1)       -- → 1/0 (currently over UI)
        local startedUi = input.touchStartedOverUI(1)  -- → 1/0 (started over UI — sticky)
    end

    -- ==== Gesture helpers (two fingers) ====
    -- Default to fingers 1 and 2; pass explicit 1-based indices for combos.
    if count >= 2 then
        local dist  = input.pinchDistance()    -- → screen px between fingers (0 if <2)
        local angle = input.pinchAngle()       -- → angle a→b in degrees (rotation gesture)
        local cx, cy = input.pinchCenter()     -- → screen midpoint between fingers

        -- Or with explicit indices: input.pinchDistance(2, 3)
    end

    -- Common pinch-zoom pattern:
    --   local lastDist = nil
    --   function OnTick()
    --     if input.touchCount() == 2 then
    --       local d = input.pinchDistance()
    --       if lastDist then zoom(d - lastDist) end  -- delta>0 → spreading
    --       lastDist = d
    --     else lastDist = nil end
    --   end

    -- ==== Keyboard ====
    local held = input.key("space")            -- → 1/0 (key held down)
    local pressed = input.keyDown("w")         -- → 1/0 (key pressed this frame)
    -- Supported key names:
    -- "a"-"z", "0"-"9", "space", "left", "right", "up", "down",
    -- "left shift", "right shift", "left ctrl", "return", "escape",
    -- "tab", "backspace"
end

    local items = spawn.getItems()             -- → array of "name|alias" strings

    local count = spawn.getItemCount()         -- → count

    local savesStr = spawn.getSavesString("\n")-- → string joined by separator
    local saveCount = spawn.getSaveCount()     -- → count

    -- === Built-in resource saves catalog ===
    local rSaves = spawn.getResourceSaves()          -- → array of "name|id" strings
    local rSavesStr = spawn.getResourceSavesString() -- → string joined by separator
    local rSaveCount = spawn.getResourceSaveCount()  -- → count

    -- === Mods catalog ===
    local mods = spawn.getMods()               -- → array of "name|id" strings
    local modsStr = spawn.getModsString("\n")  -- → string joined by separator
    local modCount = spawn.getModCount()       -- → count

    -- === Creating objects (deferred → OnSpawned) ===
    -- OnSpawned(requestId, entities) — entities is ALWAYS an Entity array
    local req1 = spawn.create("crate_wood", 0, 5)       -- alias, x, y → requestId
    local req2 = spawn.createWithAngle("crate_wood", 0, 5, 90) -- alias, x, y, angle → requestId

    -- === Cloning ===
    local req3 = spawn.clone(1, 5, 5)         -- entityId, x, y → requestId (persistent)
    local req4 = spawn.cloneTemp(1, 5, 5)     -- entityId, x, y → requestId (temporary, not saved)

    -- === Spawning saves and mods (OnSpawned receives all entities from the save) ===
    local req5 = spawn.createSave("my_save", 0, 5)  -- name, x, y → requestId
    local req6 = spawn.createMod("Cool Mod", 3, 5)  -- name, x, y → requestId

    -- === Destroy (immediate) ===
    spawn.destroy(1)                          -- entityId

    -- === Lookup by alias ===
    local name = spawn.getNameByAlias("crate_wood")   -- → localized name by alias
    local exists = spawn.existsByAlias("crate_wood")  -- → 1/0 (does alias exist)
end

-- ============================================================
-- chip — VPChip / LuaChip introspection (11 methods)
-- ============================================================
-- Works for VPChip and LuaChip mechanics. UIControl / standard mechanics
-- are handled by their own modules (uicontrol / mechanic).

function DemoChip()
    local id = inputs.entity.target

    -- Discovery
    local has = chip.has(id)                    -- → 1/0 (is it a chip at all?)
    local kind = chip.getType(id)               -- → "VPChip" / "LuaChip" / nil

    -- I/O lists — array of "key|GateDataType" strings (Number, String, Vector, Entity, etc)
    local ins = chip.getInputs(id)              -- → array of "key|type"
    local outs = chip.getOutputs(id)            -- → array of "key|type"

    -- Read / Write a gate by name (looks up outputs first, then inputs)
    local val = chip.getValue(id, "score")      -- → value (number/string/vec/...) or nil
    local ok = chip.setValue(id, "score", 42)   -- → 1/0 (0 if the gate is wired)

    -- Wire check (true if input gate has a connection)
    local wired = chip.hasWire(id, "score")     -- → 1/0

    -- Activation gate
    local act = chip.getActivation(id)          -- → current ActivationInput value
    chip.setActivation(id, 1)                  -- → 1/0 (set ActivationInput)

    -- Meta
    local name = chip.getName(id)               -- → visual name / __name input
    local tps = chip.getTPS(id)                 -- → ticks per second
end

-- ============================================================
-- signal — event bus between chips
-- ============================================================

function DemoSignal()
    -- Subscribe (in OnInit or body)
    local subId = signal.on("damage", function(data)
        print("damage: " .. data.amount)       -- data — any Lua type
    end)

    -- Emit (instant — every subscriber is called right away)
    signal.emit("damage", { amount = 10, source = "fire" })

    -- Deferred emit (fires at the start of next tick via __flush_signals)
    signal.defer("game_over", { winner = "player1" })

    -- Unsubscribe
    signal.off("damage", subId)
end

-- ============================================================
-- register_module / require — user modules across chips
-- ============================================================

function DemoModules()
    -- One chip registers a module (in OnInit or body)
    register_module("math_utils", {
        clamp = function(v, lo, hi)
            return math.max(lo, math.min(hi, v))
        end,
        lerp = function(a, b, t)
            return a + (b - a) * t
        end,
        distance = function(x1, y1, x2, y2)
            return math.sqrt((x2 - x1)^2 + (y2 - y1)^2)
        end,
    })

    -- Another chip imports it (at any time after registration)
    local utils = require("math_utils")
    local clamped = utils.clamp(150, 0, 100)   -- → 100
    local mid = utils.lerp(0, 10, 0.5)          -- → 5
    local dist = utils.distance(0, 0, 3, 4)     -- → 5

    -- A module can hold any Lua values: functions, tables, numbers
    -- Stored in _G.__modules — available to every chip
end

-- ============================================================
-- debug — print / warn / error_log (3 globals)
-- ============================================================
-- All three write to the Lua console AND Unity's CustomDebug log,
-- but with different severity levels (visible in console filters).

function DemoDebug()
    -- Info (white) — diagnostic / regular trace
    print("hello")                             -- → console + Debug.Log
    print("x =", 42, "y =", 3.14)             -- multiple args joined by tab
    print(type(Entity(1)))                     -- → "table"

    -- Warning (yellow) — recoverable issue, attention needed
    warn("config not loaded, using defaults")
    warn("low health:", inputs.num.hp)

    -- Error (red) — actual bug / unrecoverable problem
    error_log("missing required input: target_entity")
    error_log("invalid value:", inputs.num.action, "(expected 0..3)")

    -- Compile / runtime errors are also auto-logged by the engine:
    -- [compile] ERROR: ...
    -- [runtime] ERROR in OnTick: ...
end
